
import React, { useState, useEffect, useCallback } from 'react';
import { Plus, PieChart, Activity, CheckCircle2, Clock, Loader2, X, Check, Play, FileBarChart, ChevronRight, ChevronDown, Copy } from 'lucide-react';
import { Theme, Brand, ModuleType } from '../types';
import { useBrandCatalog } from '../hooks/useBrandCatalog';
import { tasksAPI } from '../api/tasks';
import { knowledgeAPI, ExtractionTask } from '../api/knowledge';
import Pagination from './Pagination';

interface AnalyzeListProps {
  theme: Theme;
  onNewTask: (productId?: string | null) => void;
  currentBrand: Brand;
  onNavigate: (module: ModuleType, taskId?: string) => void;
  initialProduct?: string | null;
}

interface BatchItem {
  batchId: string;
  batchName: string;
  taskCount: number;
  completedCount: number;
  status: 'completed' | 'processing';
  targetBrand: string;
  platforms: string[];
  totalQuestions: number;
  createdAt: string;
  completedAt: string | null;
}

const PAGE_SIZE = 10;

const AnalyzeList: React.FC<AnalyzeListProps> = ({ theme, onNewTask, currentBrand, onNavigate, initialProduct = null }) => {
  const isDark = theme === 'dark';
  const { catalog } = useBrandCatalog(currentBrand);
  const [selectedProduct, setSelectedProduct] = useState<string>(initialProduct || '');
  
  const allProducts = catalog.flatMap(cat => cat.products);
  const [batches, setBatches] = useState<BatchItem[]>([]);
  const [isLoadingTasks, setIsLoadingTasks] = useState(false);
  const [tasksError, setTasksError] = useState<string | null>(null);
  const [currentPage, setCurrentPage] = useState(1);
  const [totalBatches, setTotalBatches] = useState(0);

  const [expandedBatches, setExpandedBatches] = useState<Set<string>>(new Set());
  const [batchSubTasks, setBatchSubTasks] = useState<Record<string, any[]>>({});
  const [loadingSubTasks, setLoadingSubTasks] = useState<Set<string>>(new Set());
  const [syncingBatches, setSyncingBatches] = useState<Set<string>>(new Set());
  const [copiedTaskId, setCopiedTaskId] = useState<string | null>(null);

  // 鍒涘缓浠诲姟妯℃€佹鐘舵€?  const [isModalOpen, setIsModalOpen] = useState(false);
  const [selectedPlatforms, setSelectedPlatforms] = useState<string[]>(['doubao']);  // 鏀逛负澶氶€?  const [selectedExtractionTask, setSelectedExtractionTask] = useState<string | null>(null);
  const [extractionTasks, setExtractionTasks] = useState<ExtractionTask[]>([]);
  const [inputBrandName, setInputBrandName] = useState('');
  const [isLoadingExtractionTasks, setIsLoadingExtractionTasks] = useState(false);
  const [isCreatingTask, setIsCreatingTask] = useState(false);
  const [createError, setCreateError] = useState<string | null>(null);
  const loadTasks = useCallback(async (skipOverride?: number) => {
    setIsLoadingTasks(true);
    setTasksError(null);
    try {
      const skip = skipOverride !== undefined ? skipOverride : (currentPage - 1) * PAGE_SIZE;
      const response = await tasksAPI.listBatches(undefined, skip, PAGE_SIZE);
      setTotalBatches(response.total || 0);

      if (!response.batches || response.batches.length === 0) {
        setBatches([]);
        return;
      }

      const items: BatchItem[] = response.batches.map((b: any) => {
        const date = b.created_at;
        const formattedDate = date ? new Date(date).toLocaleString('zh-CN', {
          year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit'
        }).replace(/\//g, '-') : '';
        const completedDate = b.completed_at ? new Date(b.completed_at).toLocaleString('zh-CN', {
          year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit'
        }).replace(/\//g, '-') : null;

        return {
          batchId: b.batch_id,
          batchName: b.batch_name || b.batch_id,
          taskCount: b.task_count || 0,
          completedCount: b.completed_count || 0,
          status: b.status,
          targetBrand: b.target_brand || '',
          platforms: b.platforms || [],
          totalQuestions: b.total_questions || 0,
          createdAt: formattedDate,
          completedAt: completedDate,
        };
      });

      setBatches(items);
    } catch (error) {
      setTasksError(error instanceof Error ? error.message : '鍔犺浇澶辫触');
    } finally {
      setIsLoadingTasks(false);
    }
  }, [currentPage]);

  useEffect(() => { loadTasks(); }, [loadTasks]);

  const toggleBatch = useCallback(async (batchId: string) => {
    setExpandedBatches(prev => {
      const next = new Set(prev);
      if (next.has(batchId)) {
        next.delete(batchId);
      } else {
        next.add(batchId);
        if (!batchSubTasks[batchId]) {
          setLoadingSubTasks(p => new Set(p).add(batchId));
          tasksAPI.getBatchTasks(batchId).then((resp: any) => {
            const tasks = Array.isArray(resp) ? resp : [];
            setBatchSubTasks(p => ({ ...p, [batchId]: tasks }));
          }).catch(() => {
            setBatchSubTasks(p => ({ ...p, [batchId]: [] }));
          }).finally(() => {
            setLoadingSubTasks(p => { const n = new Set(p); n.delete(batchId); return n; });
          });
        }
      }
      return next;
    });
  }, [batchSubTasks]);

  const handleSyncBatch = useCallback(async (batchId: string) => {
    setSyncingBatches(p => new Set(p).add(batchId));
    try {
      const result = await tasksAPI.syncBatchProgress(batchId);
      if (result?.tasks) {
        setBatchSubTasks(p => ({ ...p, [batchId]: result.tasks }));
      }
      if (!expandedBatches.has(batchId)) {
        setExpandedBatches(p => new Set(p).add(batchId));
      }
      await loadTasks();
    } catch (e) {
      console.error('鍚屾杩涘害澶辫触:', e);
    } finally {
      setSyncingBatches(p => { const n = new Set(p); n.delete(batchId); return n; });
    }
  }, [expandedBatches, loadTasks]);

  // 鎵撳紑妯℃€佹鏃跺姞杞借瘝鏉℃彁鐐间换鍔?  useEffect(() => {
    if (isModalOpen) {
      loadExtractionTasks();
    }
  }, [isModalOpen, currentBrand.name]);

  // 鍔犺浇璇嶆潯鎻愮偧浠诲姟
  const loadExtractionTasks = async () => {
    setIsLoadingExtractionTasks(true);
    try {
      const response = await knowledgeAPI.getExtractionHistory({
        brand_name: currentBrand.name,
        limit: 20  // 鏄剧ず鏈€鏂扮殑20涓?      });
      setExtractionTasks(response.tasks || []);
    } catch (error) {
      console.error('鍔犺浇璇嶆潯鎻愮偧浠诲姟澶辫触:', error);
      setExtractionTasks([]);
    } finally {
      setIsLoadingExtractionTasks(false);
    }
  };

  const getPackType = (modelType?: string): 'normal' | 'semantic' | 'manual' => {
    if (modelType === 'traditional' || modelType === 'word_expand') return 'normal';
    if (modelType === 'manual') return 'manual';
    return 'semantic';
  };

  // 鍏抽棴妯℃€佹
  const handleCloseModal = () => {
    setIsModalOpen(false);
    setSelectedPlatforms(['doubao']);
    setSelectedExtractionTask(null);
    setCreateError(null);
    setInputBrandName('');
  };

  // 鍒涘缓鐜扮姸鍒嗘瀽浠诲姟
  const handleCreateTask = async () => {
    if (!inputBrandName.trim()) {
      setCreateError('璇疯緭鍏ュ搧鐗屽悕绉?);
      return;
    }
    if (!selectedExtractionTask) {
      setCreateError('璇烽€夋嫨璇嶅寘');
      return;
    }
    if (selectedPlatforms.length === 0) {
      setCreateError('璇疯嚦灏戦€夋嫨涓€涓狝I骞冲彴');
      return;
    }

    const selectedExtTask = extractionTasks.find(t => t.id === selectedExtractionTask);
    const packType = getPackType(selectedExtTask?.model_type);
    if (packType === 'semantic') {
      setCreateError('璇箟璇嶅寘鏆備笉鏀寔锛岃閫夋嫨鏅€氳瘝鍖呮垨鎵嬪姩瀵煎叆璇嶅寘');
      return;
    }

    setIsCreatingTask(true);
    setCreateError(null);

    try {
      const keywordsResponse = await knowledgeAPI.getExtractionTaskKeywords(selectedExtractionTask!);
      
      let keywords: string[] = [];
      
      if (keywordsResponse.model_type === 'traditional') {
        const positive = keywordsResponse.positive_questions || [];
        const negative = keywordsResponse.negative_questions || [];
        const brand = keywordsResponse.brand_questions || [];
        keywords = [...positive, ...negative, ...brand];
      } else if (keywordsResponse.model_type === 'word_expand' || keywordsResponse.model_type === 'manual') {
        const longtail = keywordsResponse.longtail_questions?.map(q => 
          typeof q === 'string' ? q : q.text || q.source_keyword || ''
        ) || [];
        keywords = longtail;
        if (keywords.length === 0 && keywordsResponse.keywords) {
          keywords = keywordsResponse.keywords.map(kw => 
            typeof kw === 'string' ? kw : kw.text || ''
          );
        }
      } else {
        const longtail = keywordsResponse.longtail_questions?.map(q => 
          typeof q === 'string' ? q : q.text || q.source_keyword || ''
        ) || [];
        const semantic = keywordsResponse.semantic_extension_questions?.map(q => 
          typeof q === 'string' ? q : q.text || q.source_keyword || ''
        ) || [];
        keywords = [...longtail, ...semantic];
        if (keywords.length === 0 && keywordsResponse.keywords) {
          keywords = keywordsResponse.keywords.map(kw => 
            typeof kw === 'string' ? kw : kw.text || ''
          );
        }
      }

      if (keywords.length === 0) {
        setCreateError('鎵€閫夎瘝鍖呮病鏈夊叧閿瘝');
        setIsCreatingTask(false);
        return;
      }

      // 鏅€氳瘝鍖呭彧澶勭悊绗竴鏉?      if (packType === 'normal') {
        keywords = [keywords[0]];
      }

      const result = await tasksAPI.createTaskFromKeywords({
        keywords: keywords,
        platforms: selectedPlatforms,
        target_brand: inputBrandName.trim(),
        keyword_extraction_task_id: selectedExtractionTask!,
        brand_name: inputBrandName.trim(),
        product_name: undefined,
        keyword_pack_type: packType,
      });

      console.log('鉁?浠诲姟鍒涘缓鎴愬姛:', result.task_id, '鎬婚棶棰樻暟:', result.total_questions);

      // 3. 鍒涘缓鎴愬姛锛屽埛鏂板垪琛紙绗?椤碉級
      setCurrentPage(1);
      await loadTasks(0);
      handleCloseModal();
    } catch (error) {
      console.error('鍒涘缓浠诲姟澶辫触:', error);
      setCreateError(error instanceof Error ? error.message : '鍒涘缓鐜扮姸鍒嗘瀽浠诲姟澶辫触');
    } finally {
      setIsCreatingTask(false);
    }
  };


  const platformNames: Record<string, string> = {
    'doubao': '璞嗗寘',
    'deepseek': 'DeepSeek',
    'wenxin': '鏂囧績涓€瑷€',
    'qianwen': '閫氫箟鍗冮棶',
    'yuanbao': '鑵捐鍏冨疂',
    'kimi': 'kimi'
  };

  return (
    <div className={`h-full flex flex-col ${isDark ? 'bg-geo-bg' : 'bg-white'}`}>
      {/* Main Content Area */}
      <div className={`flex-1 flex flex-col overflow-hidden relative h-full ${isDark ? 'bg-geo-card' : 'bg-white'}`}>
        <div className="flex-1 p-8 lg:p-12 overflow-y-auto no-scrollbar">
          <div className="max-w-[1400px] mx-auto space-y-10">
            <div className="flex justify-between items-end">
              <div>
                <h2 className={`text-4xl font-semibold tracking-tight ${isDark ? 'text-geo-text-main' : 'text-slate-900'}`}>鐜扮姸鍒嗘瀽浠诲姟</h2>
              </div>
              <button 
                onClick={() => { setInputBrandName(''); setIsModalOpen(true); }}
                className={`flex items-center gap-3 px-5 py-2.5 rounded-2xl font-semibold text-sm  shadow-sm hover-scale
                  ${isDark ? 'bg-geo-blue text-white shadow-blue-glow hover:bg-blue-500' : 'bg-blue-500 text-white shadow-blue-500/30 hover:bg-blue-600'}
                `}
              >
                <Plus className="w-5 h-5" /> 鍒涘缓鍒嗘瀽浠诲姟
              </button>
            </div>

            {tasksError && (
              <div className={`p-4 rounded-xl mb-4 ${isDark ? 'bg-red-900/20 border border-red-800 text-red-400' : 'bg-red-50 border border-red-200 text-red-600'}`}>
                <div className="flex items-center justify-between">
                  <span>{tasksError}</span>
                  <button
                    onClick={loadTasks}
                    className={`px-4 py-2 rounded-lg text-xs font-bold ${isDark ? 'bg-red-800 text-white hover:bg-red-700' : 'bg-red-100 text-red-700 hover:bg-red-200'}`}
                  >
                    閲嶈瘯
                  </button>
                </div>
              </div>
            )}

            {isLoadingTasks ? (
              <div className="flex items-center justify-center py-20">
                <Loader2 className="w-6 h-6 animate-spin text-geo-blue" />
              </div>
            ) : batches.length === 0 ? (
              <div className={`rounded-2xl border p-12 text-center ${isDark ? 'bg-geo-bg/30 border-geo-border' : 'bg-white border-slate-200'}`}>
                <p className="text-sm opacity-50 mb-4">鏆傛棤鍒嗘瀽璁板綍</p>
                <button
                  onClick={() => { setInputBrandName(''); setIsModalOpen(true); }}
                  className={`px-6 py-2 rounded-lg text-xs font-bold ${isDark ? 'bg-geo-blue text-white hover:bg-blue-500' : 'bg-blue-500 text-white hover:bg-blue-600 shadow-blue-500/30'}`}
                >
                  鍒涘缓棣栦釜鍒嗘瀽浠诲姟
                </button>
              </div>
            ) : (
              <div className="space-y-4">
                <div className={`rounded-2xl border overflow-hidden ${isDark ? 'border-geo-border' : 'border-slate-200'}`}>
                  <table className="w-full text-sm">
                    <thead>
                      <tr className={isDark ? 'bg-geo-card/80 text-geo-text-sec' : 'bg-slate-50 text-slate-500'}>
                        <th className="text-left px-4 py-3 font-semibold w-8"></th>
                        <th className="text-left px-4 py-3 font-semibold">鎵规</th>
                        <th className="text-left px-4 py-3 font-semibold">骞冲彴</th>
                        <th className="text-left px-4 py-3 font-semibold">闂鏁?/th>
                        <th className="text-left px-4 py-3 font-semibold">鐘舵€?/th>
                        <th className="text-left px-4 py-3 font-semibold">鍒涘缓鏃堕棿</th>
                        <th className="text-left px-4 py-3 font-semibold">鎿嶄綔</th>
                      </tr>
                    </thead>
                    <tbody>
                      {batches.map((batch) => {
                        const isCompleted = batch.status === 'completed';
                        const hasCompleted = (batch.completedCount || 0) > 0;
                        const isExpanded = expandedBatches.has(batch.batchId);
                        const subTasks = batchSubTasks[batch.batchId] || [];
                        const isLoadingSub = loadingSubTasks.has(batch.batchId);

                        return (
                          <React.Fragment key={batch.batchId}>
                            {/* 鎵规涓昏 */}
                            <tr
                              className={`cursor-pointer transition-colors border-t ${
                                isDark
                                  ? 'border-geo-border hover:bg-geo-bg/50'
                                  : 'border-slate-100 hover:bg-slate-50'
                              } ${isExpanded ? (isDark ? 'bg-geo-bg/30' : 'bg-slate-50/50') : ''}`}
                              onClick={() => toggleBatch(batch.batchId)}
                            >
                              <td className="px-4 py-3">
                                {isExpanded
                                  ? <ChevronDown className="w-4 h-4 text-geo-blue" />
                                  : <ChevronRight className={`w-4 h-4 ${isDark ? 'text-geo-text-sec' : 'text-slate-400'}`} />
                                }
                              </td>
                              <td className="px-4 py-3">
                                <div className={`font-bold truncate max-w-[280px] ${isDark ? 'text-geo-text-main' : 'text-slate-900'}`} title={batch.batchName}>
                                  {batch.batchName}
                                </div>
                                <div className={`text-xs mt-0.5 ${isDark ? 'text-geo-text-sec' : 'text-slate-400'}`}>
                                  {batch.taskCount} 涓瓙浠诲姟
                                </div>
                              </td>
                              <td className="px-4 py-3">
                                <div className="flex flex-wrap gap-1">
                                  {batch.platforms.map((p, idx) => (
                                    <span key={idx} className={`px-2 py-0.5 rounded text-xs font-semibold ${isDark ? 'bg-geo-card border border-geo-border text-geo-text-main' : 'bg-slate-100 text-slate-600'}`}>
                                      {platformNames[p] || p}
                                    </span>
                                  ))}
                                </div>
                              </td>
                              <td className={`px-4 py-3 ${isDark ? 'text-geo-text-main' : 'text-slate-700'}`}>
                                {batch.totalQuestions}
                              </td>
                              <td className="px-4 py-3">
                                <span className={`inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-xs font-semibold ${
                                  isCompleted
                                    ? (isDark ? 'bg-green-900/20 text-green-400' : 'bg-green-100 text-green-700')
                                    : (isDark ? 'bg-geo-blue/10 text-geo-blue' : 'bg-blue-100 text-blue-700')
                                }`}>
                                  {isCompleted
                                    ? <><CheckCircle2 className="w-3 h-3" /> 宸插畬鎴?/>
                                    : <><Activity className="w-3 h-3 animate-spin" /> {batch.completedCount}/{batch.taskCount}</>
                                  }
                                </span>
                              </td>
                              <td className={`px-4 py-3 text-xs ${isDark ? 'text-geo-text-sec' : 'text-slate-500'}`}>
                                {batch.createdAt}
                              </td>
                              <td className="px-4 py-3" onClick={e => e.stopPropagation()}>
                                <div className="flex items-center gap-2">
                                  <button
                                    onClick={() => handleSyncBatch(batch.batchId)}
                                    disabled={syncingBatches.has(batch.batchId)}
                                    className={`px-3 py-1.5 rounded-lg text-xs font-bold flex items-center gap-1 transition-colors ${
                                      syncingBatches.has(batch.batchId) ? 'opacity-50 cursor-not-allowed' : ''
                                    } ${isDark ? 'bg-geo-blue/10 text-geo-blue hover:bg-geo-blue/20' : 'bg-blue-50 text-blue-600 hover:bg-blue-100'}`}
                                  >
                                    {syncingBatches.has(batch.batchId)
                                      ? <><Loader2 className="w-3.5 h-3.5 animate-spin" /> 鍚屾涓?..</>
                                      : <><Activity className="w-3.5 h-3.5" /> 鍚屾杩涘害</>
                                    }
                                  </button>
                                  {hasCompleted && (
                                    <button
                                      onClick={() => onNavigate(ModuleType.DIAGNOSIS_REPORT, batch.batchId)}
                                      className={`px-3 py-1.5 rounded-lg text-xs font-bold flex items-center gap-1 transition-colors ${
                                        isDark ? 'bg-blue-500/10 text-blue-400 hover:bg-blue-500/20' : 'bg-blue-50 text-blue-600 hover:bg-blue-100'
                                      }`}
                                    >
                                      <FileBarChart className="w-3.5 h-3.5" /> 璇婃柇鎶ュ憡
                                    </button>
                                  )}
                                </div>
                              </td>
                            </tr>

                            {/* 灞曞紑鐨勫瓙浠诲姟琛?*/}
                            {isExpanded && (
                              isLoadingSub ? (
                                <tr>
                                  <td colSpan={7} className={`px-4 py-6 text-center ${isDark ? 'bg-geo-bg/20' : 'bg-slate-50/50'}`}>
                                    <Loader2 className="w-4 h-4 animate-spin text-geo-blue inline-block mr-2" />
                                    <span className={`text-xs ${isDark ? 'text-geo-text-sec' : 'text-slate-400'}`}>鍔犺浇瀛愪换鍔?..</span>
                                  </td>
                                </tr>
                              ) : subTasks.length === 0 ? (
                                <tr>
                                  <td colSpan={7} className={`px-4 py-4 text-center text-xs ${isDark ? 'bg-geo-bg/20 text-geo-text-sec' : 'bg-slate-50/50 text-slate-400'}`}>
                                    鏆傛棤瀛愪换鍔℃暟鎹?                                  </td>
                                </tr>
                              ) : (
                                <>
                                  {/* 瀛愪换鍔¤〃澶?*/}
                                  <tr className={isDark ? 'bg-geo-bg/40' : 'bg-slate-50'}>
                                    <td></td>
                                    <td className={`px-4 py-2 text-xs font-semibold ${isDark ? 'text-geo-text-sec' : 'text-slate-400'}`}>浠诲姟 ID</td>
                                    <td className={`px-4 py-2 text-xs font-semibold ${isDark ? 'text-geo-text-sec' : 'text-slate-400'}`}>骞冲彴</td>
                                    <td className={`px-4 py-2 text-xs font-semibold ${isDark ? 'text-geo-text-sec' : 'text-slate-400'}`}>杩涘害</td>
                                    <td className={`px-4 py-2 text-xs font-semibold ${isDark ? 'text-geo-text-sec' : 'text-slate-400'}`}>鐘舵€?/td>
                                    <td className={`px-4 py-2 text-xs font-semibold ${isDark ? 'text-geo-text-sec' : 'text-slate-400'}`}>鏃堕棿</td>
                                    <td className={`px-4 py-2 text-xs font-semibold ${isDark ? 'text-geo-text-sec' : 'text-slate-400'}`}>鎿嶄綔</td>
                                  </tr>
                                  {subTasks.map((task: any) => {
                                    const taskCompleted = task.status === 'completed' || task.third_party_status === 'completed';
                                    const taskDate = task.created_at ? new Date(task.created_at).toLocaleString('zh-CN', {
                                      month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit'
                                    }).replace(/\//g, '-') : '';
                                    return (
                                      <tr
                                        key={task.task_id}
                                        className={`border-t ${isDark ? 'border-geo-border/50 bg-geo-bg/20' : 'border-slate-100 bg-white'}`}
                                      >
                                        <td></td>
                                        <td className={`px-4 py-2.5 text-xs font-mono ${isDark ? 'text-geo-text-main' : 'text-slate-700'}`}>
                                          {task.task_id?.substring(0, 8)}...
                                        </td>
                                        <td className={`px-4 py-2.5`}>
                                          <span className={`px-2 py-0.5 rounded text-xs font-semibold ${isDark ? 'bg-geo-card border border-geo-border text-geo-text-main' : 'bg-slate-100 text-slate-600'}`}>
                                            {platformNames[task.platform] || task.platform}
                                          </span>
                                        </td>
                                        <td className={`px-4 py-2.5 text-xs ${isDark ? 'text-geo-text-main' : 'text-slate-700'}`}>
                                          {task.processed_questions || 0}/{task.total_questions || 0}
                                        </td>
                                        <td className="px-4 py-2.5">
                                          <span className={`inline-flex items-center gap-1 text-xs font-semibold ${
                                            taskCompleted
                                              ? (isDark ? 'text-green-400' : 'text-green-600')
                                              : (isDark ? 'text-geo-blue' : 'text-blue-600')
                                          }`}>
                                            {taskCompleted
                                              ? <><CheckCircle2 className="w-3 h-3" /> 宸插畬鎴?/>
                                              : <><Loader2 className="w-3 h-3 animate-spin" /> 鍒嗘瀽涓?/>
                                            }
                                          </span>
                                        </td>
                                        <td className={`px-4 py-2.5 text-xs ${isDark ? 'text-geo-text-sec' : 'text-slate-500'}`}>
                                          {taskDate}
                                        </td>
                                        <td className="px-4 py-2.5">
                                          {task.third_party_task_id ? (
                                            <button
                                              onClick={(e) => {
                                                e.stopPropagation();
                                                navigator.clipboard?.writeText(task.third_party_task_id).then(() => {
                                                  setCopiedTaskId(task.task_id);
                                                  setTimeout(() => setCopiedTaskId(null), 1500);
                                                }).catch(() => {});
                                              }}
                                              title={`澶嶅埗涓夋柟浠诲姟ID: ${task.third_party_task_id}`}
                                              className={`px-2 py-1 rounded text-xs font-bold flex items-center gap-1 transition-colors ${
                                                isDark ? 'text-geo-blue hover:bg-geo-blue/10' : 'text-blue-600 hover:bg-blue-50'
                                              }`}
                                            >
                                              <Copy className="w-3 h-3" /> {copiedTaskId === task.task_id ? '宸插鍒? : '澶嶅埗'}
                                            </button>
                                          ) : (
                                            <span className={`text-xs ${isDark ? 'text-geo-text-sec opacity-50' : 'text-slate-400'}`}>-</span>
                                          )}
                                        </td>
                                      </tr>
                                    );
                                  })}
                                </>
                              )
                            )}
                          </React.Fragment>
                        );
                      })}
                    </tbody>
                  </table>
                </div>
                <Pagination currentPage={currentPage} total={totalBatches} pageSize={PAGE_SIZE} onPageChange={setCurrentPage} isDark={isDark} />
              </div>
            )}
          </div>
        </div>
      </div>

        {/* Create Task Modal */}
        {isModalOpen && (
          <div className="fixed inset-0 z-50 overflow-hidden pointer-events-none">
            <div className="absolute inset-0 bg-black/40 backdrop-blur-sm pointer-events-auto transition-opacity duration-500" onClick={handleCloseModal}></div>
            <div className="absolute inset-0 flex items-center justify-center p-4 pointer-events-none">
              <div 
                className={`relative w-full max-w-2xl max-h-[90vh] overflow-y-auto rounded-2xl border shadow-sm pointer-events-auto flex flex-col ${isDark ? 'bg-zinc-900 border-white/10 text-white' : 'bg-white border-slate-200 text-slate-900'}`}
                onClick={e => e.stopPropagation()}
              >
                <header className="p-6 border-b flex items-center justify-between shrink-0">
                  <div>
                    <h3 className={`text-2xl font-semibold tracking-tight ${isDark ? 'text-white' : 'text-slate-900'}`}>
                      鍒涘缓鐜扮姸鍒嗘瀽浠诲姟
                    </h3>
                    <p className={`text-xs font-bold  opacity-50 mt-1 tracking-widest`}>
                      閫夋嫨AI骞冲彴鍜屽叧閿瘝鐢熸垚浠诲姟
                    </p>
                  </div>
                  <button onClick={handleCloseModal} className="p-2 rounded-full hover:bg-black/5 transition-colors">
                    <X className="w-5 h-5" />
                  </button>
                </header>
                
                <div className="flex-1 overflow-y-auto p-8 space-y-8">
                  <div className="space-y-8">
                    <div className="space-y-4">
                      <label className={`text-xs font-semibold  ${isDark ? 'text-zinc-500' : 'text-slate-400'}`}>鍝佺墝鍚嶇О</label>
                      <input
                        type="text"
                        value={inputBrandName}
                        onChange={(e) => setInputBrandName(e.target.value)}
                        placeholder="璇疯緭鍏ュ搧鐗屽悕绉帮紝濡傦細椋炲埄娴?(Philips)"
                        className={`w-full px-4 py-3 rounded-xl border-2 text-sm font-medium transition-all outline-none ${
                          isDark 
                            ? 'bg-zinc-800 border-zinc-700 text-white placeholder:text-zinc-600 focus:border-geo-blue' 
                            : 'bg-white border-slate-200 text-slate-900 placeholder:text-slate-400 focus:border-blue-500'
                        }`}
                      />
                    </div>

                    <div className="space-y-4">
                      <label className={`text-xs font-semibold  ${isDark ? 'text-zinc-500' : 'text-slate-400'}`}>閫夋嫨AI骞冲彴锛堝彲澶氶€夛級</label>
                      <div className="grid grid-cols-2 gap-4">
                        {['doubao', 'deepseek', 'wenxin', 'qianwen', 'yuanbao', 'kimi'].map((platform) => {
                          const platformNames: Record<string, string> = {
                            'doubao': '璞嗗寘',
                            'deepseek': 'DeepSeek',
                            'wenxin': '鏂囧績涓€瑷€',
                            'qianwen': '閫氫箟鍗冮棶',
                            'yuanbao': '鑵捐鍏冨疂',
                            'kimi': 'kimi'
                          };
                          const isSelected = selectedPlatforms.includes(platform);
                          return (
                            <button
                              key={platform}
                              onClick={() => {
                                if (isSelected) {
                                  setSelectedPlatforms(selectedPlatforms.filter(p => p !== platform));
                                } else {
                                  setSelectedPlatforms([...selectedPlatforms, platform]);
                                }
                              }}
                              className={`p-6 rounded-2xl border-2 transition-all ${isSelected
                                ? (isDark ? 'border-geo-blue bg-geo-blue/10 shadow-blue-glow' : 'border-blue-500 bg-blue-50')
                                : (isDark ? 'border-zinc-700 bg-zinc-800 hover:bg-zinc-700' : 'border-slate-200 bg-white hover:bg-slate-50')
                              }`}
                            >
                              <div className="flex items-center justify-between">
                                <div className={`text-lg font-semibold ${isDark ? 'text-white' : 'text-slate-900'}`}>{platformNames[platform]}</div>
                                {isSelected && (
                                  <div className={`w-5 h-5 rounded-full flex items-center justify-center ${isDark ? 'bg-geo-blue' : 'bg-blue-500'}`}>
                                    <Check className="w-3 h-3 text-white" />
                                  </div>
                                )}
                              </div>
                            </button>
                          );
                        })}
                      </div>
                      {selectedPlatforms.length === 0 && (
                        <p className={`text-xs ${isDark ? 'text-red-400' : 'text-red-600'}`}>璇疯嚦灏戦€夋嫨涓€涓钩鍙?/p>
                      )}
                    </div>

                    <div className="space-y-4">
                      <label className={`text-xs font-semibold  ${isDark ? 'text-zinc-500' : 'text-slate-400'}`}>閫夋嫨璇嶅寘</label>
                      {isLoadingExtractionTasks ? (
                        <div className="flex items-center justify-center py-8">
                          <Loader2 className="w-6 h-6 animate-spin text-geo-blue" />
                        </div>
                      ) : extractionTasks.length === 0 ? (
                        <div className={`p-4 rounded-2xl border-2 text-center ${isDark ? 'bg-zinc-800 border-zinc-700 text-zinc-400' : 'bg-slate-50 border-slate-200 text-slate-500'}`}>
                          鏆傛棤璇嶅寘锛岃鍏堝湪"鍏抽敭璇嶇敓鎴?涓垱寤?                        </div>
                      ) : (
                        <div className="grid grid-cols-1 gap-3 max-h-60 overflow-y-auto">
                          {extractionTasks.filter(t => getPackType(t.model_type) !== 'semantic').map((task) => {
                            const packType = getPackType(task.model_type);
                            return (
                              <button
                                key={task.id}
                                onClick={() => setSelectedExtractionTask(task.id)}
                                className={`p-4 rounded-2xl border-2 text-left transition-all ${selectedExtractionTask === task.id
                                  ? (isDark ? 'bg-geo-blue border-geo-blue text-white' : 'bg-blue-500 border-blue-500 text-white')
                                  : (isDark ? 'bg-zinc-800 border-zinc-700 text-white hover:border-geo-blue' : 'bg-white border-slate-200 text-slate-900 hover:border-blue-500')
                                }`}
                              >
                                <div className="flex items-center justify-between mb-1">
                                  <div className="flex items-center gap-2">
                                    <div className="font-bold text-sm">{task.name || task.keyword || '鏈煡鍏抽敭璇?}</div>
                                    <span className={`text-[10px] px-1.5 py-0.5 rounded font-semibold ${
                                      packType === 'manual'
                                        ? (selectedExtractionTask === task.id ? 'bg-white/20' : isDark ? 'bg-amber-500/20 text-amber-400' : 'bg-amber-100 text-amber-700')
                                        : (selectedExtractionTask === task.id ? 'bg-white/20' : isDark ? 'bg-blue-500/20 text-blue-400' : 'bg-blue-100 text-blue-700')
                                    }`}>
                                      {packType === 'manual' ? '鎵嬪姩瀵煎叆' : '鏅€氳瘝鍖?}
                                    </span>
                                  </div>
                                  <span className={`text-xs px-2 py-0.5 rounded ${selectedExtractionTask === task.id ? 'bg-white/20' : isDark ? 'bg-white/10' : 'bg-black/10'}`}>
                                    {packType === 'normal' ? '鍙栫1鏉? : `鍏ㄩ儴 ${task.count || 0} 鏉}
                                  </span>
                                </div>
                                <div className="text-xs opacity-60 mt-1">{task.date}</div>
                              </button>
                            );
                          })}
                        </div>
                      )}
                    </div>
                  </div>
                </div>

                <footer className="p-6 border-t flex justify-between items-center gap-3 shrink-0">
                  {createError && (
                    <div className={`px-4 py-2 rounded-lg text-sm ${isDark ? 'bg-red-900/20 text-red-400 border border-red-500/20' : 'bg-red-50 text-red-600 border border-red-200'}`}>
                      {createError}
                    </div>
                  )}
                  <div className="flex gap-3 ml-auto">
                    <button 
                      onClick={handleCloseModal} 
                      className={`px-6 py-3 rounded-xl text-xs font-semibold  border transition-all ${isDark ? 'bg-zinc-800 border-zinc-700 hover:text-white' : 'bg-slate-100 border-slate-200'}`}
                    >
                      鍙栨秷
                    </button>
                    <button 
                      onClick={handleCreateTask}
                      disabled={!inputBrandName.trim() || !selectedExtractionTask || selectedPlatforms.length === 0 || isCreatingTask}
                      className={`px-6 py-3 rounded-xl text-xs font-semibold  bg-blue-500 text-white shadow-blue-500/30 hover:bg-blue-600 transition-all flex items-center gap-2 ${isCreatingTask ? 'opacity-50 cursor-not-allowed' : ''}`}>
                      {isCreatingTask ? (
                        <>
                          <Loader2 className="w-4 h-4 animate-spin text-blue-500" /> 鍒涘缓涓?..
                        </>
                      ) : (
                        <>
                          <Play className="w-4 h-4 fill-current" /> 鍒涘缓鍒嗘瀽浠诲姟
                        </>
                      )}
                    </button>
                  </div>
                </footer>
              </div>
            </div>
          </div>
        )}
    </div>
  );
};

export default AnalyzeList;
